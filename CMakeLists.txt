cmake_minimum_required(VERSION 3.25)
project(lazysf2 LANGUAGES CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "SF2 is a 32-bit game: configure with -A Win32 (use the x86 preset).")
endif()

# Static CRT, to match the x86-windows-static vcpkg triplet and avoid runtime DLL dependencies.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

find_package(minhook CONFIG REQUIRED)
find_package(unofficial-enet CONFIG REQUIRED)
find_package(miniz CONFIG REQUIRED)

# The Tacview add-on (tacview/addon/LazySF2Patch/main.lua) is embedded in the DLL as a generated header. The text is split
# into pieces under MSVC's 16 KB limit for one string literal.
set(LAZYSF2_ADDON_LUA "${CMAKE_SOURCE_DIR}/tacview/addon/LazySF2Patch/main.lua")
set(LAZYSF2_ADDON_HEADER "${CMAKE_BINARY_DIR}/generated/tacview_addon_lua.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LAZYSF2_ADDON_LUA}")
file(READ "${LAZYSF2_ADDON_LUA}" LAZYSF2_ADDON_TEXT)
string(LENGTH "${LAZYSF2_ADDON_TEXT}" LAZYSF2_ADDON_LEN)
set(LAZYSF2_ADDON_BODY "")
set(LAZYSF2_ADDON_POS 0)
while(LAZYSF2_ADDON_POS LESS LAZYSF2_ADDON_LEN)
    string(SUBSTRING "${LAZYSF2_ADDON_TEXT}" ${LAZYSF2_ADDON_POS} 6000 LAZYSF2_ADDON_PIECE)
    string(APPEND LAZYSF2_ADDON_BODY "    R\"LZ(${LAZYSF2_ADDON_PIECE})LZ\"
")
    math(EXPR LAZYSF2_ADDON_POS "${LAZYSF2_ADDON_POS} + 6000")
endwhile()
file(WRITE "${LAZYSF2_ADDON_HEADER}.tmp"
     "#pragma once
// Generated by CMake from tacview/addon/LazySF2Patch/main.lua. Do not edit.
inline constexpr const char kTacviewAddonLua[] =
${LAZYSF2_ADDON_BODY};
")
file(COPY_FILE "${LAZYSF2_ADDON_HEADER}.tmp" "${LAZYSF2_ADDON_HEADER}" ONLY_IF_DIFFERENT)
file(REMOVE "${LAZYSF2_ADDON_HEADER}.tmp")

    src/hooks/pilot_twist.cpp
    src/hooks/menu_hooks.cpp
    src/hooks/options_gameplay_splice.cpp
    src/hooks/aircraft_track.cpp
    src/hooks/weapon_track.cpp
    src/hooks/tacview_recorder.cpp
    src/hooks/terrain_export.cpp
    src/hooks/game_files.cpp
    src/hooks/wic_jpeg.cpp
    src/hooks/world_walk.cpp
    src/acmi/acmi_format.cpp
    src/acmi/tacview_core.cpp
    src/acmi/tacview_terrain.cpp
    src/acmi/tacview_writer.cpp
    src/acmi/acmi_zip.cpp

)
set_target_properties(lazysf2 PROPERTIES OUTPUT_NAME winmm)
target_compile_features(lazysf2 PRIVATE cxx_std_20)
target_compile_definitions(lazysf2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(lazysf2 PRIVATE /W4 /permissive-)
target_include_directories(lazysf2 PRIVATE src "${CMAKE_BINARY_DIR}/generated")
# ENet is linked statically. Do NOT link winmm here: this DLL is itself winmm.dll, and ENet's three winmm imports
# are satisfied by src/net/enet_winmm_shim.cpp (they call the real system DLL).
target_link_libraries(lazysf2 PRIVATE minhook::minhook unofficial::enet::enet miniz::miniz ws2_32 delayimp shell32 ole32 uuid)
# ws2_32 is delay-loaded: it is only touched by ENet (enet_initialize, on the network thread, which only exists in
# host mode), so a player with host mode off never has it loaded into the game.
target_link_options(lazysf2 PRIVATE /DELAYLOAD:ws2_32.dll)

# The developer tools go to their own folder (build\<Config>	ools): next to the proxy they would load it as their winmm.dll.
set_target_properties(lazysf2-net-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/tools")

# The synthetic client (console, developer tool, not shipped). A normal exe, so it links winmm and ws2_32 as usual.

target_include_directories(lazysf2-bot PRIVATE src tools/bot)

set_target_properties(lazysf2-acmi PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/tools")
